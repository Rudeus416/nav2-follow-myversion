# 对应用户需求：R13 R23（原话及追溯边界见 nav2/CODE_GUIDE.md）。
# 添加/修改逻辑：声明依赖、构建/安装或插件注册；具体见本文件指令，不改系统 Nav2 源码。
# 本次仅加注释；需求关联不是精确创建/提交记录。
cmake_minimum_required(VERSION 3.16)
project(visual_car_whole_body)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(dwb_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(ompl REQUIRED)
add_library(visual_car_whole_body SHARED src/plugins.cpp)
target_include_directories(visual_car_whole_body PRIVATE ${OpenCV_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
ament_target_dependencies(visual_car_whole_body nav2_core nav2_costmap_2d dwb_core pluginlib)
target_link_libraries(visual_car_whole_body ${OpenCV_LIBRARIES} ${OMPL_LIBRARIES})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
pluginlib_export_plugin_description_file(dwb_core plugins.xml)
install(TARGETS visual_car_whole_body LIBRARY DESTINATION lib)
add_executable(whole_body_test src/test.cpp)
target_include_directories(whole_body_test PRIVATE ${OpenCV_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
ament_target_dependencies(whole_body_test nav2_core nav2_costmap_2d dwb_core pluginlib)
target_link_libraries(whole_body_test ${OpenCV_LIBRARIES} ${OMPL_LIBRARIES})
enable_testing()
add_test(NAME whole_body COMMAND whole_body_test)
ament_package()
add_executable(plugin_load_test src/load_test.cpp)
ament_target_dependencies(plugin_load_test nav2_core nav2_costmap_2d dwb_core pluginlib)
add_executable(layer_test src/layer_test.cpp)
ament_target_dependencies(layer_test nav2_core nav2_costmap_2d dwb_core pluginlib)
